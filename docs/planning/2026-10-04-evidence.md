- 搜索：真实引擎同步返回，取消计时器没有机会在扫描中执行。单次 ms 是本机合成读数，不能推断目标手机时延。
- 楼层提交：null/空串/空数组/false 均触发 invalidateFloorAt(1)；合法 0 也触发 1。没给与第 0 楼塌成同形。undefined 没触发，不能把所有非法输入一概写成同一种结果。
- 知情：三个无效值归一为 0，seenCount 按键计为已看，互动门却拒绝。这是统计与判定不一致；不能为修统计而授予未知者互动权。
- 预算：三块各为 6 字符标签＋120 字符正文，明确不属于常驻。budget=200，balanced 输出 218，relevance 471，recency 200。常驻全保留是既有设计；本探针特意排除该理由，只核预算漏计与按比例选取。
- 存储：调用真实 StorageManager，宿主替身处于 B，save('A') 把 A 写入 B 的 metadata 并 confirmed A。这证明底层边界欠保护；尚未证明真实生产调用方栅栏下可达，计划要求追完整调用时序。
- 消费矩阵：81 行来自真实 App 表；80 是目录数。32 件仅接生命周期不代表 32 个缺陷；F1–F6 本身是特定文本形态扫描口径，不等于全部产品能力。

## 探针复跑

以下源码复制到临时 `.cjs` 后运行；可用 RP_PLAN_REPO / LS_PLAN_REPO 指定仓库路径。使用合成数据，不发网络请求、不读用户存档。它报告现状而不把结果当修复后的期望值。

```javascript
const fs = require('node:fs');
const vm = require('node:vm');
const path = require('node:path');
const { pathToFileURL } = require('node:url');
const rp = process.env.RP_PLAN_REPO || '/home/user/ruby-phone';
const ls = process.env.LS_PLAN_REPO || '/home/user/lonsha-memory-plugin';
(async () => {
  const { LonShaBridge } = await import(pathToFileURL(path.join(rp, 'apps/memory/lonsha-bridge.js')));
  const touched = [];
  const bridge = new LonShaBridge({get: () => null, set: () => true}, {invalidateFloorAt: f => {touched.push(f); return 1;}});
  const invalidFloorResults = [];
  for (const v of [null, '', [], false, undefined]) {
    touched.length = 0;
    bridge.onFloorCommitted(v);
    invalidFloorResults.push({input: String(v), type: typeof v, calls: touched.slice()});
  }
  touched.length = 0;
  bridge.onFloorCommitted(0);
  const validZero = touched.slice();
  const SG = await import(pathToFileURL(path.join(rp, 'apps/socialguard/socialguard-data.js')));
  const post = SG.normalizePost({id:'p', authorId:'author', text:'x', seenBy:{A:null,B:[],C:false,D:12}}, 0).post;
  const sgRead = SG.readingsOf([post],[],100);
  const interactions = Object.fromEntries(['A','B','C','D'].map(k => [k, SG.mayInteractWith(post,k)]));
  const ctx = vm.createContext({console});
  vm.runInContext(fs.readFileSync(path.join(ls,'injection-router.js'),'utf8'),ctx);
  const R = ctx.LonShaInjectionRouter;
  const blocks = ['[非驻留一]'+ '甲'.repeat(120), '[非驻留二]'+ '乙'.repeat(120), '[非驻留三]'+ '丙'.repeat(120)];
  const budgetResults = ['balanced','relevance','recency'].map(strategy => {
    const trace = {};
    const result = R.trimToBudget(blocks.join('\n'),200,blocks,strategy,{trace});
    return {strategy,budget:200,chars:result.length,resident:trace.residentKept,kept:trace.keptIdxInAll,hardTruncated:trace.hardTruncated};
  });
  const { GlobalSearchEngine } = await import(pathToFileURL(path.join(rp,'apps/memory/global-search-engine.js')));
  const data = Array.from({length:10000},(_,i)=>({title:'item '+i,body:'命中段'+ '文'.repeat(700),ts:i}));
  const engine = new GlobalSearchEngine({sources:[{id:'seed',label:'seed',items:()=>data}]});
  let cancellationFired = false;
  const timer = setTimeout(()=>{ cancellationFired=true; },0);
  const started = performance.now();
  const result = engine.searchAll('命中',{full:true,isCancelled:()=>cancellationFired});
  const searchProbe = {isPromise:!!(result && typeof result.then==='function'),scanned:result.scanned,cancelled:result.cancelled,timerAtReturn:cancellationFired,ms:performance.now()-started};
  clearTimeout(timer);
  const {MATRIX,FACE_KEYS,faceCounts} = await import(pathToFileURL(path.join(rp,'config/app-consumption-matrix.js')));
  const lifecycleOnly = MATRIX.filter(row => row.faces.F6_lifecycle && FACE_KEYS.filter(k=>k!=='F6_lifecycle').every(k => !row.faces[k])).map(x=>x.appId);
  const O = require(path.join(ls,'memory-organs.js'));
  const metaB = {};
  global.window = {SillyTavern:{getContext:()=>({chatId:'B',chatMetadata:metaB,saveChat:async()=>true})}};
  const store = new O.StorageManager();
  const mismatchedChatSaved = await store.save('A',{summaries:[]});
  const storageProbe = {requestedChat:'A',contextChat:'B',returned:mismatchedChatSaved,metadataChat:metaB.extensions?.lonsha_memory?.chatId,confirmed:store._confirmed};
  delete global.window;
  console.log(JSON.stringify({floorCommit:{invalidFloorResults,validZero},seenBy:{normalized:post.seenBy,seenCount:sgRead.seenCount,interactions},budgetResults,searchProbe,storageProbe,matrix:{rows:MATRIX.length,counts:faceCounts(MATRIX),lifecycleOnly}},null,2));
})().catch(e=>{console.error(e.stack);process.exitCode=1;});

```
